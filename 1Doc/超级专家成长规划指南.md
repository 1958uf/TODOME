**关键代码：流式 SSE 处理（前端核心能力）**

```typescript
// hooks/useStreamChat.ts
// 功能：处理 LLM 流式输出，实时更新消息内容
// 输入：用户消息、会话ID
// 输出：流式消息状态和控制函数

import { useState, useCallback, useRef } from 'react'

interface Message {
  id: string
  role: 'user' | 'assistant'
  content: string
  thinking?: string  // 推理模型的思考过程
  isStreaming?: boolean
}

export function useStreamChat(sessionId: string) {
  const [messages, setMessages] = useState<Message[]>([])
  const [isLoading, setIsLoading] = useState(false)
  const abortRef = useRef<AbortController | null>(null)

  const sendMessage = useCallback(async (content: string) => {
    // 添加用户消息
    const userMsg: Message = { id: Date.now().toString(), role: 'user', content }
    const assistantMsg: Message = {
      id: (Date.now() + 1).toString(),
      role: 'assistant',
      content: '',
      isStreaming: true
    }

    setMessages(prev => [...prev, userMsg, assistantMsg])
    setIsLoading(true)

    abortRef.current = new AbortController()

    try {
      const response = await fetch('/api/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ session_id: sessionId, message: content }),
        signal: abortRef.current.signal
      })

      if (!response.ok) throw new Error(`HTTP ${response.status}`)

      const reader = response.body!.getReader()
      const decoder = new TextDecoder()

      while (true) {
        const { done, value } = await reader.read()
        if (done) break

        // 解析 SSE 数据块
        const chunk = decoder.decode(value)
        const lines = chunk.split('\n').filter(l => l.startsWith('data: '))

        for (const line of lines) {
          const data = line.slice(6)  // 移除 'data: '
          if (data === '[DONE]') break

          try {
            const parsed = JSON.parse(data)
            const delta = parsed.choices?.[0]?.delta?.content || ''

            setMessages(prev => prev.map(m =>
              m.id === assistantMsg.id
                ? { ...m, content: m.content + delta }
                : m
            ))
          } catch { /* 忽略非 JSON 行 */ }
        }
      }
    } catch (err: any) {
      if (err.name !== 'AbortError') {
        setMessages(prev => prev.map(m =>
          m.id === assistantMsg.id
            ? { ...m, content: '⚠️ 请求失败，请重试', isStreaming: false }
            : m
        ))
      }
    } finally {
      setMessages(prev => prev.map(m =>
        m.id === assistantMsg.id ? { ...m, isStreaming: false } : m
      ))
      setIsLoading(false)
    }
  }, [sessionId])

  const stopStream = useCallback(() => {
    abortRef.current?.abort()
  }, [])

  return { messages, isLoading, sendMessage, stopStream }
}
```

```python
    allow_origins=settings.ALLOWED_ORIGINS,
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# 请求日志中间件
@app.middleware("http")
async def log_requests(request: Request, call_next):
    """记录所有请求的方法、路径、耗时"""
    start = time.time()
    response = await call_next(request)
    duration = (time.time() - start) * 1000
```

```python
# backend/app/api/v1/chat.py
# 功能：对话接口，支持流式和非流式输出
# 输入：用户消息、会话 ID
# 输出：SSE 流式响应

from fastapi import APIRouter, Depends, HTTPException
from fastapi.responses import StreamingResponse
from pydantic import BaseModel
import asyncio
import json

from app.core.security import get_current_user
from app.agent.engine import AgentEngine
from app.db.repositories.session import SessionRepository

router = APIRouter()

class ChatRequest(BaseModel):
    """对话请求模型"""
    session_id: str
    message: str
    stream: bool = True

@router.post("/completions")
async def chat_completions(
    request: ChatRequest,
    current_user = Depends(get_current_user),
    session_repo: SessionRepository = Depends()
):
    """
    功能：处理用户对话请求，返回 SSE 流式响应
    输入：ChatRequest（含 session_id 和用户消息）
    输出：SSE 流（text/event-stream）
    """
    # 验证会话归属
    session = await session_repo.get(request.session_id)
    if not session or session.user_id != current_user.id:
        raise HTTPException(status_code=404, detail="会话不存在")

    # 初始化 Agent 引擎
    engine = AgentEngine(session_id=request.session_id, user_id=current_user.id)

    async def generate_stream():
        """SSE 流生成器"""
        try:
            async for chunk in engine.astream(request.message):
                # SSE 格式：data: <json>\n\n
                yield f"data: {json.dumps(chunk, ensure_ascii=False)}\n\n"
            yield "data: [DONE]\n\n"
        except Exception as e:
            error_chunk = {"error": str(e)}
            yield f"data: {json.dumps(error_chunk)}\n\n"

    return StreamingResponse(
        generate_stream(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "X-Accel-Buffering": "no"  # 禁用 Nginx 缓冲，确保流式传输
        }
    )
```

**LangGraph 核心概念**：

```python
# agent/graph/state.py
# 功能：定义多 Agent 系统的全局共享状态
# 输入：初始任务
# 输出：任务执行结果

from typing import TypedDict, Annotated, Sequence, Optional
from langchain_core.messages import BaseMessage
import operator

class AgentState(TypedDict):
    """
    全局 Agent 状态：所有节点共享、按需读写
    
    设计原则：
    - messages 使用 operator.add 实现 append-only（防止状态竞争）
    - 其他字段直接覆盖（最新状态优先）
    """
    # 消息历史（所有 Agent 共同可见）
    messages: Annotated[Sequence[BaseMessage], operator.add]
    
    # 当前执行的任务描述
    task: str
    
    # 规划器生成的执行步骤
    plan: list[str]
    
    # 各步骤的执行结果
    results: dict[str, str]
    
    # 下一个执行节点（由 Orchestrator 决定）
    next_node: str
    
    # 循环防护计数器
    iteration: int
    
    # 知识库检索到的上下文
    retrieved_context: str
    
    # 最终回答
    final_answer: str
    
    # 错误信息（如有）
    error: Optional[str]
```

```python
# agent/graph/builder.py
# 功能：构建企业级多 Agent 工作流图
# 输入：工具列表、LLM 配置
# 输出：编译后的可执行 LangGraph

from langgraph.graph import StateGraph, END
from langgraph.checkpoint.memory import MemorySaver
from langchain_openai import ChatOpenAI
import os

from .state import AgentState
from .nodes import (
    orchestrator_node,
    researcher_node,
    planner_node,
    executor_node,
    critic_node,
    rag_retriever_node
)
from .routers import should_continue, route_after_orchestrator

def build_agent_graph(tools: list, stream_mode: bool = True):
    """
    功能：构建完整的多 Agent 协作图
    
    图结构：
    ┌─────────────────────────────────┐
    │         orchestrator             │
    │    （分析意图，路由到专职 Agent）   │
    └──┬──────┬──────┬────────┬───────┘
       │      │      │        │
    planner researcher executor critic
       │      │      │        │
       └──────┴──────┴────────┘
              （完成后回到 orchestrator）
    
    输入：tools（工具列表），stream_mode（是否流式）
    输出：编译后的 CompiledGraph
    """
    workflow = StateGraph(AgentState)
    
    # 注册节点
    workflow.add_node("orchestrator", orchestrator_node)
    workflow.add_node("researcher",   researcher_node)
    workflow.add_node("planner",      planner_node)
    workflow.add_node("executor",     executor_node)
    workflow.add_node("critic",       critic_node)
    workflow.add_node("rag_retriever", rag_retriever_node)
    
    # 设置入口
    workflow.set_entry_point("orchestrator")
    
    # orchestrator 的条件路由
    workflow.add_conditional_edges(
        "orchestrator",
        route_after_orchestrator,
        {
            "researcher":    "researcher",
            "planner":       "planner",
            "executor":      "executor",
            "critic":        "critic",
            "rag_retriever": "rag_retriever",
            "__end__":       END
        }
    )
    
    # 专职 Agent 完成后回到 orchestrator（形成循环）
    for node in ["researcher", "planner", "executor", "rag_retriever"]:
        workflow.add_edge(node, "orchestrator")
    
    # critic 直接结束（质检是最后一步）
    workflow.add_edge("critic", END)
    
    # 使用内存 Checkpointer 实现会话持久化
    memory = MemorySaver()
    
    return workflow.compile(checkpointer=memory)
```

**ReAct 模式核心实现**：

```python
# agent/patterns/react.py
# 功能：ReAct（推理-行动-观察）模式实现
# 核心思想：Thought → Action → Observation 循环，直到任务完成

from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, SystemMessage, ToolMessage
from langchain_core.tools import BaseTool
import json
import os

REACT_SYSTEM_PROMPT = """你是一个企业级 AI 助手，使用 ReAct 方式解决问题。

工作方式：
1. Thought（思考）：分析当前状态，决定下一步行动
2. Action（行动）：调用工具获取信息或执行操作  
3. Observation（观察）：分析工具返回结果
4. 重复以上步骤，直到能给出最终答案

原则：
- 不确定事实时，一定先用工具查询，不要猜测
- 每次只调用一个最必要的工具
- 在 Final Answer 之前，总结所有观察结果
"""

class ReActAgent:
    """
    功能：基于 ReAct 模式的单 Agent 实现
    适用场景：需要多步骤工具调用的任务
    """
    
    MAX_ITERATIONS = 8
    
    def __init__(self, tools: list[BaseTool], model: str = "meta/llama-3.1-70b-instruct"):
        self.tools = {t.name: t for t in tools}
        self.llm = ChatOpenAI(
            base_url=os.getenv("NVIDIA_BASE_URL"),
            api_key=os.getenv("NVIDIA_API_KEY"),
            model=model,
            temperature=0.2
        ).bind_tools(tools)  # 绑定工具，启用 Function Calling
        self.trace = []  # 记录完整推理轨迹（可观察性）
    
    async def arun(self, task: str) -> dict:
        """
        功能：异步执行 ReAct 推理循环
        输入：task（任务描述）
        输出：{"answer": str, "trace": list, "tool_calls": int}
        """
        messages = [
            SystemMessage(content=REACT_SYSTEM_PROMPT),
            HumanMessage(content=task)
        ]
        
        total_tool_calls = 0
        
        for i in range(self.MAX_ITERATIONS):
            response = await self.llm.ainvoke(messages)
            
            # 没有工具调用 = 最终回答
            if not response.tool_calls:
                self.trace.append({"step": i, "type": "final_answer", "content": response.content})
                return {
                    "answer": response.content,
                    "trace": self.trace,
                    "tool_calls": total_tool_calls
                }
            
            # 执行工具调用
            messages.append(response)  # 添加 assistant 消息（含 tool_calls）
            
            for tool_call in response.tool_calls:
                tool_name = tool_call["name"]
                tool_args = tool_call["args"]
                
                self.trace.append({
                    "step": i,
                    "type": "tool_call",
                    "tool": tool_name,
                    "args": tool_args
                })
                
                # 执行工具
                if tool_name in self.tools:
                    try:
                        result = await self.tools[tool_name].arun(tool_args)
                        total_tool_calls += 1
                    except Exception as e:
                        result = f"工具执行失败: {str(e)}"
                else:
                    result = f"未知工具: {tool_name}"
                
                # 将工具结果加入消息链
                messages.append(ToolMessage(
                    content=str(result),
                    tool_call_id=tool_call["id"]
                ))
                
                self.trace.append({"step": i, "type": "observation", "content": str(result)[:200]})
        
        return {
            "answer": "达到最大迭代次数，任务未完成",
            "trace": self.trace,
            "tool_calls": total_tool_calls
        }
```

**RAG 全流程实现**：

```python
# agent/rag/pipeline.py
# 功能：完整 RAG 流水线，包含文档处理、向量化、检索、重排序
# 输入：文档路径 / 查询文本
# 输出：知识库上下文片段

from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_community.document_loaders import PyPDFLoader, UnstructuredMarkdownLoader
from langchain_community.vectorstores.pgvector import PGVector
from langchain_openai import OpenAIEmbeddings
from langchain.retrievers import ContextualCompressionRetriever
from langchain.retrievers.document_compressors import CrossEncoderReranker
from langchain_community.cross_encoders import HuggingFaceCrossEncoder
import os

class RAGPipeline:
    """
    功能：企业级 RAG 流水线，支持多格式文档、混合检索、重排序
    
    性能优化点：
    1. 文档分块：递归字符分块 + 语义边界感知
    2. 混合检索：向量检索（语义）+ BM25（关键词）融合
    3. 重排序：Cross-Encoder 精排，提升 Top-K 准确性
    4. 上下文压缩：只提取最相关段落，控制 Token 消耗
    """
    
    CHUNK_SIZE    = 512    # 经验最优值：大模型 context = 2K 时的推荐分块
    CHUNK_OVERLAP = 64     # 10-15% overlap 防止跨块语义丢失
    TOP_K         = 10     # 初始检索数量（重排序前）
    RERANK_TOP_K  = 4      # 重排序后保留数量（注入 Prompt 的上下文）
    
    def __init__(self, collection_name: str):
        """
        功能：初始化 RAG 流水线，连接向量数据库
        输入：collection_name（知识库名称，对应 PGVector 的 collection）
        """
        self.collection_name = collection_name
        
        # Embedding 模型（使用 NVIDIA NIM 的 embedding API 或本地 BGE）
        self.embeddings = OpenAIEmbeddings(
            base_url=os.getenv("NVIDIA_BASE_URL"),
            api_key=os.getenv("NVIDIA_API_KEY"),
            model="nvidia/nv-embedqa-e5-v5"  # NVIDIA 推荐的中英文 embedding 模型
        )
        
        # 向量数据库（PGVector，与 PostgreSQL 集成）
        self.vectorstore = PGVector(
            connection_string=os.getenv("DATABASE_URL"),
            collection_name=collection_name,
            embedding_function=self.embeddings
        )
        
        # 文本分块器
        self.splitter = RecursiveCharacterTextSplitter(
            chunk_size=self.CHUNK_SIZE,
            chunk_overlap=self.CHUNK_OVERLAP,
            separators=["\n\n", "\n", "。", "！", "？", " ", ""]
        )
    
    def ingest(self, filepath: str, metadata: dict = {}) -> int:
        """
        功能：导入文档到知识库（支持 PDF/Markdown/TXT）
        输入：filepath（文件路径），metadata（文档元数据）
        输出：导入的文档块数量
        """
        # 选择合适的 Loader
        if filepath.endswith(".pdf"):
            loader = PyPDFLoader(filepath)
        elif filepath.endswith((".md", ".markdown")):
            loader = UnstructuredMarkdownLoader(filepath)
        else:
            from langchain_community.document_loaders import TextLoader
            loader = TextLoader(filepath, encoding="utf-8")
        
        documents = loader.load()
        
        # 注入元数据
        for doc in documents:
            doc.metadata.update(metadata)
            doc.metadata["source"] = os.path.basename(filepath)
        
        # 分块
        chunks = self.splitter.split_documents(documents)
        
        # 向量化并存储
        self.vectorstore.add_documents(chunks)
        
        return len(chunks)
    
    def retrieve(self, query: str) -> str:
        """
        功能：检索与 query 最相关的知识块，经重排序后拼接为上下文
        输入：query（用户问题或 Agent 子任务描述）
        输出：格式化的知识库上下文字符串
        """
        # 基础向量检索
        base_retriever = self.vectorstore.as_retriever(
            search_type="mmr",  # Maximum Marginal Relevance，避免重复
            search_kwargs={"k": self.TOP_K, "fetch_k": 20}
        )
        
        # 重排序（Cross-Encoder 精排）
        reranker = HuggingFaceCrossEncoder(model_name="BAAI/bge-reranker-base")
        compressor = CrossEncoderReranker(model=reranker, top_n=self.RERANK_TOP_K)
        
        retriever = ContextualCompressionRetriever(
            base_compressor=compressor,
            base_retriever=base_retriever
        )
        
        docs = retriever.get_relevant_documents(query)
        
        if not docs:
            return ""
        
        # 格式化为 LLM 可读的上下文
        context_parts = ["以下是从知识库检索到的相关信息：\n"]
        for i, doc in enumerate(docs, 1):
            source = doc.metadata.get("source", "未知来源")
            context_parts.append(f"[{i}] 来源：{source}\n{doc.page_content}\n")
        
        return "\n".join(context_parts)
```

```python
# core/cache.py
# 功能：Redis 缓存管理，实现多级缓存策略
# 输入：缓存 key、数据
# 输出：缓存读写结果

import redis.asyncio as aioredis
import json
from typing import Optional, Any
import os

class CacheManager:
    """
    功能：统一缓存管理，实现会话缓存、限流、分布式锁
    
    缓存策略设计：
    - 会话消息：TTL 24h（活跃会话）
    - 用户信息：TTL 1h（变化不频繁）
    - LLM 响应：TTL 1h（相同 prompt 的缓存）
    - 限流计数：TTL 1min（滑动窗口）
    """
    
    SESSION_TTL = 86400    # 24 小时
    USER_TTL    = 3600     # 1 小时
    LLM_TTL     = 3600     # 1 小时
    
    def __init__(self):
        self.redis = aioredis.from_url(
            os.getenv("REDIS_URL", "redis://localhost:6379"),
            decode_responses=True,
            max_connections=20
        )
    
    async def get_session_context(self, session_id: str) -> Optional[list]:
        """
        功能：从 Redis 获取会话上下文（最近 N 轮消息）
        输入：session_id
        输出：消息列表或 None
        """
        key = f"session:context:{session_id}"
        data = await self.redis.get(key)
        return json.loads(data) if data else None
    
    async def set_session_context(self, session_id: str, context: list):
        """缓存会话上下文，刷新 TTL"""
        key = f"session:context:{session_id}"
        await self.redis.setex(key, self.SESSION_TTL, json.dumps(context, ensure_ascii=False))
    
    async def check_rate_limit(self, user_id: str, limit: int = 60) -> bool:
        """
        功能：基于滑动窗口的 API 限流（每用户每分钟 N 次）
        输入：user_id，limit（限额）
        输出：True=允许，False=超限
        """
        key = f"ratelimit:{user_id}"
        pipe = self.redis.pipeline()
        pipe.incr(key)
        pipe.expire(key, 60)  # 1 分钟窗口
        results = await pipe.execute()
        return results[0] <= limit
    
    async def acquire_lock(self, resource: str, ttl: int = 30) -> bool:
        """
        功能：分布式锁（防止并发重复处理同一个 Agent 任务）
        输入：resource（锁名称），ttl（超时秒数）
        输出：True=获取成功，False=已被锁定
        """
        key = f"lock:{resource}"
        result = await self.redis.set(key, "1", nx=True, ex=ttl)
        return result is True
    
    async def release_lock(self, resource: str):
        """释放分布式锁"""
        await self.redis.delete(f"lock:{resource}")
```

```python
# core/llm/client.py
# 功能：统一 LLM 调用客户端，支持多模型、流式输出、Token 管理、降级重试
# 输入：消息列表、模型配置
# 输出：LLM 响应（流式或完整）

from openai import AsyncOpenAI
from typing import AsyncGenerator, Optional
import tiktoken
import time
import os
import structlog

logger = structlog.get_logger()

class LLMClient:
    """
    功能：生产级 LLM 调用封装，解决以下工程问题：
    1. Token 超限：自动压缩历史消息
    2. 响应延迟：超时控制 + 重试
    3. 成本控制：Token 计数和记录
    4. 可观察性：调用日志和 Langfuse 追踪
    5. 多模型：NVIDIA NIM 主路 + 降级备用
    """
    
    # 各模型的上下文窗口大小
    CONTEXT_WINDOWS = {
        "meta/llama-3.1-70b-instruct": 128000,
        "meta/llama-3.1-8b-instruct":  128000,
        "nvidia/nemotron-3-nano-omni-30b-a3b-reasoning": 32000,
        "qwen/qwen2.5-72b-instruct":   128000,
    }
    
    # 安全留给输出的 Token 比例
    OUTPUT_RESERVE_RATIO = 0.25
    
    def __init__(self):
        self.client = AsyncOpenAI(
            base_url=os.getenv("NVIDIA_BASE_URL", "https://integrate.api.nvidia.com/v1"),
            api_key=os.getenv("NVIDIA_API_KEY"),
            timeout=60.0
        )
        self.encoder = tiktoken.get_encoding("cl100k_base")
    
    def count_tokens(self, text: str) -> int:
        """估算文本的 Token 数量"""
        return len(self.encoder.encode(text))
    
    def compress_messages(self, messages: list, model: str, system_prompt: str) -> list:
        """
        功能：当消息超出上下文窗口时，压缩历史消息
        策略：保留 system prompt + 最近 N 轮 + 必要摘要
        输入：消息列表，目标模型，系统提示词
        输出：压缩后的消息列表
        """
        max_context = self.CONTEXT_WINDOWS.get(model, 8192)
        max_input = int(max_context * (1 - self.OUTPUT_RESERVE_RATIO))
        
        # 计算 system prompt 的 token
        system_tokens = self.count_tokens(system_prompt)
        available = max_input - system_tokens - 500  # 保留 buffer
        
        compressed = []
        total_tokens = 0
        
        # 从最新消息向前追加，直到填满可用 token
        for msg in reversed(messages):
            msg_tokens = self.count_tokens(msg.get("content", ""))
            if total_tokens + msg_tokens > available:
                break
            compressed.insert(0, msg)
            total_tokens += msg_tokens
        
        # 如果丢弃了历史消息，在开头加一条摘要说明
        if len(compressed) < len(messages):
            omitted = len(messages) - len(compressed)
            compressed.insert(0, {
                "role": "system",
                "content": f"（注：为控制长度，已省略最早的 {omitted} 条历史消息）"
            })
        
        return compressed
    
    async def astream(
        self,
        messages: list,
        model: str = "meta/llama-3.1-70b-instruct",
        temperature: float = 0.3,
        tools: Optional[list] = None,
        session_id: Optional[str] = None
    ) -> AsyncGenerator[str, None]:
        """
        功能：异步流式调用 LLM，自动处理 Token 超限和错误重试
        输入：messages（消息列表）、model、temperature、tools
        输出：AsyncGenerator，逐 chunk yield 文字内容
        """
        start_time = time.time()
        total_tokens = 0
        
        kwargs = {
            "model": model,
            "messages": messages,
            "temperature": temperature,
            "stream": True,
            "max_tokens": 4096,
        }
        if tools:
            kwargs["tools"] = tools
        
        try:
            stream = await self.client.chat.completions.create(**kwargs)
            
            async for chunk in stream:
                choice = chunk.choices[0] if chunk.choices else None
                if not choice:
                    continue
                
                delta = choice.delta
                
                # 工具调用 chunk
                if delta.tool_calls:
                    yield {"type": "tool_call", "data": delta.tool_calls}
                
                # 文字内容 chunk
                if delta.content:
                    total_tokens += 1  # 粗略估算
                    yield {"type": "content", "data": delta.content}
                
                # 结束
                if choice.finish_reason:
                    yield {"type": "done", "finish_reason": choice.finish_reason}
        
        except Exception as e:
            latency = (time.time() - start_time) * 1000
            logger.error("llm_stream_error",
                        model=model,
                        error=str(e),
                        latency_ms=round(latency, 2),
                        session_id=session_id)
            yield {"type": "error", "data": str(e)}
        
        finally:
            latency = (time.time() - start_time) * 1000
            logger.info("llm_call_complete",
                       model=model,
                       latency_ms=round(latency, 2),
                       approx_tokens=total_tokens,
                       session_id=session_id)
```

## 7. 面试高频考点与回答框架

### 7.1 技术面必问题

```
Q1: 讲一下你理解的 RAG 和 Fine-tuning 的区别，什么场景选哪个？
框架：从"知识更新频率"、"成本"、"可解释性"三个维度分析
回答：
  知识频繁更新 → RAG（如文档库、产品手册）
  需要行为/风格改变 → Fine-tuning（如品牌语气、专业术语）
  兼具两者 → RAG + Fine-tuning（最佳效果，但成本高）
  举例：我在 XX 项目中选择 RAG 的原因是知识库每周更新，Fine-tuning 迭代成本太高

Q2: 描述一下你实现的多 Agent 系统架构，怎么解决 Agent 间的协调问题？
框架：从"通信协议"、"状态管理"、"循环防护"三个维度回答
回答：
  用 LangGraph StateGraph 管理全局状态，避免直接 Agent-to-Agent 通信
  Orchestrator Agent 负责路由，专职 Agent 只做自己的事
  通过 iteration 计数器 + 相似状态检测防止无限循环
  具体：我在 TODOME V6 中...（结合项目经验）

Q3: 你如何优化 LLM 的 Token 消耗和响应延迟？
框架：分"请求优化"和"提示词优化"两层
回答：
  请求层：流式输出（用户感知延迟下降 70%）、上下文压缩（滑动窗口 + 摘要）、缓存相同 prompt 的响应
  提示词层：Few-shot 例子精简、System Prompt 结构化、避免重复指令
  监控：用 Langfuse 追踪每次调用的 Token 消耗，找出最贵的调用链

Q4: 讲一下向量数据库的检索原理，你在项目中如何优化检索准确率？
框架：从"索引算法"、"相似度计算"、"工程优化"三层回答
回答：
  原理：HNSW 近似最近邻算法，在精度和速度间平衡
  优化链路：文档分块策略（chunk size/overlap）→ embedding 模型选择 → MMR 去重 → Cross-Encoder 重排序
  数据：优化前 Top-3 准确率 XX%，优化后 XX%

Q5: 如何保障 Agent 系统在高并发下的稳定性？
框架：从"限流"、"降级"、"监控"三个维度
回答：
  限流：Redis 滑动窗口限流（用户级 + 全局级）
  降级：LLM 调用超时自动重试（小模型降级）、工具调用失败的 fallback 逻辑
  监控：Prometheus 采集 API 延迟/错误率，Grafana 告警，Langfuse 追踪 LLM 调用
```
